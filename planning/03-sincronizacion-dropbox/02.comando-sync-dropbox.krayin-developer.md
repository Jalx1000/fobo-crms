Schema::create('smd_synced_events', function (Blueprint $table) {
    $table->id();
    $table->string('external_id')->unique()->comment('_id del evento en SMD');
    $table->string('source_file')->comment('Ruta del archivo en Dropbox');
    $table->string('source')->default('dropbox')->comment('dropbox|webhook');
    $table->unsignedBigInteger('activity_id')->nullable()->comment('Activity creada en CRM');
    $table->unsignedBigInteger('lead_id')->nullable();

    $table->timestamp('processed_at')->useCurrent();
    $table->timestamps();

    $table->index('external_id');
    $table->index('processed_at');
});

### Comando

```php
<?php

namespace Webkul\Admin\Console\Commands;

use Illuminate\Console\Command;
use Illuminate\Support\Facades\Http;
use Illuminate\Support\Facades\Log;
use Illuminate\Support\Facades\DB;
use Webkul\Admin\Services\IncomingAppointmentService;

class SyncDropboxAppointments extends Command
{
    protected $signature   = 'smd:sync-dropbox
                              {--date= : Fecha a sincronizar (Y-m-d). Default: hoy y ayer}
                              {--dry-run : Solo lista archivos sin procesar}';
    protected $description = 'Sincroniza citas confirmadas en SMD desde la carpeta Dropbox compartida';

    public function __construct(
        protected IncomingAppointmentService $incomingService
    ) {
        parent::__construct();
    }

    public function handle(): int
    {
        $token  = config('services.dropbox.access_token');
        $folder = config('services.dropbox.appointments_folder', '/appointments');

        if (! $token) {
            $this->error('DROPBOX_ACCESS_TOKEN no configurado');
            return self::FAILURE;
        }

        $dates = $this->option('date')
            ? [$this->option('date')]
            : [now()->toDateString(), now()->subDay()->toDateString()];

        $processed = 0;
        $skipped   = 0;
        $errors    = 0;

        foreach ($dates as $date) {
            $files = $this->listDropboxFiles("{$folder}/{$date}", $token);

            foreach ($files as $file) {
                $externalId = pathinfo($file['name'], PATHINFO_FILENAME);

                // Idempotencia: skip si ya fue procesado
                if (DB::table('smd_synced_events')->where('external_id', $externalId)->exists()) {
                    $skipped++;
                    continue;
                }

                if ($this->option('dry-run')) {
                    $this->line("  [DRY-RUN] {$file['path_lower']}");
                    continue;
                }

                $json = $this->downloadDropboxFile($file['path_lower'], $token);

                if (! $json) {
                    $errors++;
                    continue;
                }

                try {
                    $result = $this->incomingService->processDropbox($json);

                    DB::table('smd_synced_events')->insert([
                        'external_id'  => $externalId,
                        'source_file'  => $file['path_lower'],
                        'source'       => 'dropbox',
                        'activity_id'  => $result['activity_id'] ?? null,
                        'lead_id'      => $result['lead_id']     ?? null,
                        'raw_payload'  => json_encode($json),
                        'processed_at' => now(),
                        'created_at'   => now(),
                        'updated_at'   => now(),
                    ]);

                    $processed++;
                    $this->info("  ✓ Procesado: {$externalId}");
                } catch (\Exception $e) {
                    $errors++;
                    Log::error('[SyncDropbox] Error procesando evento', [
                        'external_id' => $externalId,
                        'error'       => $e->getMessage(),
                    ]);
                    $this->error("  ✗ Error en {$externalId}: {$e->getMessage()}");
                }
            }
        }

        $this->info("Resumen: {$processed} procesados, {$skipped} ya existían, {$errors} errores.");
        return $errors > 0 ? self::FAILURE : self::SUCCESS;
    }

    private function listDropboxFiles(string $path, string $token): array
    {
        $response = Http::withToken($token)
            ->withHeaders(['Content-Type' => 'application/json'])
            ->post('https://api.dropboxapi.com/2/files/list_folder', ['path' => $path]);

        if (! $response->successful()) {
            Log::warning('[SyncDropbox] No se pudo listar carpeta', ['path' => $path, 'status' => $response->status()]);
            return [];
        }

        return collect($response->json('entries', []))
            ->filter(fn($e) => $e['.tag'] === 'file' && str_ends_with($e['name'], '.json'))
            ->values()
            ->toArray();
    }

    private function downloadDropboxFile(string $path, string $token): ?array
    {
        $response = Http::withToken($token)
            ->withHeaders(['Dropbox-API-Arg' => json_encode(['path' => $path])])
            ->post('https://content.dropboxapi.com/2/files/download');

        if (! $response->successful()) {
            Log::error('[SyncDropbox] No se pudo descargar archivo', ['path' => $path]);
            return null;
        }

        return $response->json();
    }
}
```

## Criterios de aceptación

- [ ] `php artisan smd:sync-dropbox --dry-run` lista archivos sin crear registros
- [ ] `php artisan smd:sync-dropbox --date=2026-05-19` procesa solo esa fecha
- [ ] Ejecutar dos veces la misma fecha no duplica registros (idempotencia)
- [ ] Los errores en un archivo no detienen el procesamiento de los siguientes
- [ ] La tabla `smd_synced_events` registra todos los eventos procesados
- [ ] El comando aparece en `php artisan schedule:list`
