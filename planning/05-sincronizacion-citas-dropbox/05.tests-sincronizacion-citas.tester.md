## Test 1 — Evento nuevo: crea Lead, Activity y Person correctamente

```php
it('crea Lead, Activity y Person para un evento nuevo de Dropbox', function () {
    $service = app(IncomingAppointmentService::class);
    $data    = smdEventFixture();

    $result = $service->processDropbox($data);

    expect($result)->toHaveKeys(['lead_id', 'activity_id']);

    // Activity creada con fechas correctas
    $activity = Activity::find($result['activity_id']);
    expect($activity)->not->toBeNull();
    expect($activity->schedule_from)->toBe('2026-05-12 15:45:00');
    expect($activity->schedule_to)->toBe('2026-05-12 16:15:00');
    expect($activity->title)->toBe('Aclaramiento');

    // Person (paciente) con nombre y teléfono correctos
    $lead   = Lead::find($result['lead_id']);
    $person = $lead->person;
    expect($person->name)->toBe('Abad Rojas');

    // El teléfono del paciente debe estar en contact_numbers
    $phones = collect($person->contact_numbers)->pluck('value');
    expect($phones)->toContain('60000001');

    // Doctor asignado a la Activity
    $doctors = $activity->doctors;
    expect($doctors)->not->toBeEmpty();
    expect($doctors->first()->unique_id)->toBe('6525b4de28cb7c0008b77dbb');
});
```

## Test 2 — Evento actualizado: actualiza Activity sin crear duplicados

```php
it('actualiza la Activity existente cuando llega el mismo _id con datos modificados', function () {
    $service = app(IncomingAppointmentService::class);
    $data    = smdEventFixture();

    // Primera vez: crear
    $first = $service->processDropbox($data);

    // Registrar en smd_synced_events (como haría el comando)
    DB::table('smd_synced_events')->insert([
        'external_id'  => $data['_id'],
        'source_file'  => '/smd-events/2026-05-12/event.json',
        'source'       => 'dropbox',
        'activity_id'  => $first['activity_id'],
        'lead_id'      => $first['lead_id'],
        'raw_payload'  => json_encode($data),
        'processed_at' => now(),
        'created_at'   => now(),
        'updated_at'   => now(),
    ]);

    // Segunda vez: mismo _id, fechas distintas
    $updated = smdEventFixture([
        'startDate' => '2026-05-12T16:00:00.000Z',
        'endDate'   => '2026-05-12T16:30:00.000Z',
        'summary'   => 'Aclaramiento Actualizado',
    ]);
    $existing = DB::table('smd_synced_events')->where('external_id', $data['_id'])->first();
    $second   = $service->updateDropbox($updated, $existing);

    // Mismo activity_id (no se creó uno nuevo)
    expect($second['activity_id'])->toBe($first['activity_id']);
    expect($second['action'])->toBe('updated');

    // Las fechas de la Activity se actualizaron
    $activity = Activity::find($first['activity_id']);
    expect($activity->schedule_from)->toBe('2026-05-12 16:00:00');
    expect($activity->title)->toBe('Aclaramiento Actualizado');

    // No se crearon duplicados
    expect(Activity::count())->toBe(1);
    expect(Lead::count())->toBe(1);
});
```

## Test 3 — Evento archivado: cancela el Lead

```php
it('cancela el Lead cuando llega un evento con archived=true', function () {
    $service = app(IncomingAppointmentService::class);
    $data    = smdEventFixture();

    // Crear primero
    $first = $service->processDropbox($data);

    $existing = (object) [
        'activity_id' => $first['activity_id'],
        'lead_id'     => $first['lead_id'],
    ];

    // Archivar
    $result = $service->cancelDropbox($data['_id'], $existing);

    expect($result['action'])->toBe('cancelled');

    // Lead debe estar cerrado
    $lead = Lead::find($first['lead_id']);
    expect($lead->status)->toBe(0);
    expect($lead->lost_reason)->toContain('SMD');
    expect($lead->closed_at)->not->toBeNull();

    // Activity NO fue eliminada (queda como registro histórico)
    expect(Activity::find($first['activity_id']))->not->toBeNull();
});
```

## Test 4 — Doctor no encontrado: se crea automáticamente

```php
it('crea un Doctor nuevo cuando owner._id no existe en el CRM', function () {
    $service = app(IncomingAppointmentService::class);

    // Asegurar que no existe el doctor
    expect(Doctor::where('unique_id', '6525b4de28cb7c0008b77dbb')->count())->toBe(0);

    $data   = smdEventFixture();
    $result = $service->processDropbox($data);

    $doctor = Doctor::where('unique_id', '6525b4de28cb7c0008b77dbb')->first();
    expect($doctor)->not->toBeNull();
    expect($doctor->name)->toBe('Nuevo Medico');
    expect($doctor->phone)->toBe('76123456');
    expect($doctor->is_active)->toBeTrue();
});

it('hace backfill de unique_id cuando el doctor existe por telefono', function () {
    // Doctor creado manualmente con teléfono, sin unique_id
    $doctor = Doctor::create([
        'name'      => 'Dr. Existente',
        'phone'     => '76123456',
        'is_active' => true,
    ]);

    $service = app(IncomingAppointmentService::class);
    $data    = smdEventFixture([
        'owner' => [
            '_id'      => '6525b4de28cb7c0008b77dbb',
            'name'     => 'Nuevo',
            'lastName' => 'Medico',
            'phone'    => '76123456',
        ],
    ]);

    $result = $service->processDropbox($data);

    // No se creó un doctor duplicado
    expect(Doctor::count())->toBe(1);

    // Se hizo backfill del unique_id
    $doctor->refresh();
    expect($doctor->unique_id)->toBe('6525b4de28cb7c0008b77dbb');

    // La Activity se asignó al doctor existente
    $activity = Activity::find($result['activity_id']);
    expect($activity->doctors->first()->id)->toBe($doctor->id);
});
```

it('crea una Person nueva cuando el paciente no existe en el CRM', function () {

    expect(Person::count())->toBe(0);

        'contact_numbers' => [['value' => '60000001', 'label' => 'work']],
        'emails'          => [['value' => '60000001@whatsapp.sofopolis.net', 'label' => 'work']],

    $service = app(IncomingAppointmentService::class);

    // El Lead apunta a la persona existente
    $lead = Lead::find($result['lead_id']);
