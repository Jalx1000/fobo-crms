## Casos de prueba requeridos

### Grupo 1 — `searchPatient()`

```php
it('retorna paciente encontrado por teléfono', function () {
    Http::fake([
        '*/patients*' => Http::response([
            'data' => [['_id' => 'smd-abc123', 'fullName' => 'Juan Pérez', 'phone' => '77788990']]
        ], 200),
    ]);

    $service = app(ShareMeDataService::class);
    $result  = $service->searchPatient('77788990');

    expect($result)->not->toBeEmpty();
    expect($result[0]['_id'])->toBe('smd-abc123');
});

it('retorna array vacío si no encuentra paciente', function () {
    Http::fake(['*/patients*' => Http::response(['data' => []], 200)]);

    $result = app(ShareMeDataService::class)->searchPatient('00000000');
    expect($result)->toBeEmpty();
});

it('retorna array vacío si SMD falla', function () {
    Http::fake(['*/patients*' => Http::response([], 500)]);

    $result = app(ShareMeDataService::class)->searchPatient('77788990');
    expect($result)->toBeEmpty();
});
```

### Grupo 2 — `createPatient()`

```php
it('crea paciente exitosamente en SMD', function () {
    Http::fake([
        '*/patients/user*' => Http::response(['_id' => 'new-smd-id', 'fullName' => 'Ana López'], 201),
    ]);

    $result = app(ShareMeDataService::class)->createPatient([
        'first_name' => 'Ana',
        'last_name'  => 'López',
        'phone'      => '71234567',
    ]);

    expect($result['success'])->toBeTrue();
    expect($result['data']['_id'])->toBe('new-smd-id');
});

it('detecta paciente duplicado en SMD', function () {
    Http::fake([
        '*/patients/user*' => Http::response(
            ['message' => '@error.duplicatePatient'],
            400
        ),
    ]);

    $result = app(ShareMeDataService::class)->createPatient([
        'first_name' => 'Ana', 'last_name' => 'López', 'phone' => '71234567',
    ]);

    expect($result['success'])->toBeFalse();
    expect($result['duplicate'])->toBeTrue();
});
```

### Grupo 3 — `updatePatient()`

it('actualiza paciente existente en SMD', function () {

    // ... ejecutar AppointmentService::process()

    expect($person->fresh()->smd_patient_id)->toBe('dup-smd-id');
});
