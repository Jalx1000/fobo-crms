## Objetivo

```php
<?php

namespace Webkul\Admin\Jobs;

use Illuminate\Bus\Queueable;
use Illuminate\Contracts\Queue\ShouldQueue;
use Illuminate\Foundation\Bus\Dispatchable;
use Illuminate\Queue\InteractsWithQueue;
use Illuminate\Queue\SerializesModels;
use Illuminate\Support\Facades\DB;
use Illuminate\Support\Facades\Log;
use Webkul\Admin\Services\DropboxService;
use Webkul\Admin\Services\IncomingAppointmentService;
use Webkul\Admin\Services\SmdStatusMapper;

class ProcessDropboxNotification implements ShouldQueue
{
    use Dispatchable, InteractsWithQueue, Queueable, SerializesModels;

    public int $tries   = 3;
    public int $timeout = 60;

    public function handle(
        DropboxService $dropbox,
        IncomingAppointmentService $incoming,
        SmdStatusMapper $mapper,
    ): void {
        $cursor = $this->getCursor();

        if (! $cursor) {
            $cursor = $dropbox->getInitialCursor(
                config('smd.dropbox.appointments_folder', '/smd-events')
            );

            if (! $cursor) {
                Log::error('[Dropbox Webhook Job] No se pudo obtener cursor inicial');
                return;
            }

            $this->saveCursor($cursor);
            Log::info('[Dropbox Webhook Job] Cursor inicializado');
            return;  // no hay archivos "nuevos" en la primera inicialización
        }

        $result = $dropbox->listChanges($cursor);

        if ($result === null) {
            Log::error('[Dropbox Webhook Job] Error llamando list_folder/continue');
            return;
        }

        $newCursor = $result['cursor'];
        $entries   = $result['entries'];

        Log::info('[Dropbox Webhook Job] Cambios recibidos', ['count' => count($entries)]);

        foreach ($entries as $entry) {
            // Solo archivos JSON (no carpetas, no eliminaciones)
            if (($entry['.tag'] ?? '') !== 'file') {
                continue;
            }

            if (! str_ends_with($entry['name'] ?? '', '.json')) {
                continue;
            }

            $this->processFile($entry, $incoming, $dropbox, $mapper);
        }

        $this->saveCursor($newCursor);
    }

    private function processFile(
        array $entry,
        IncomingAppointmentService $incoming,
        DropboxService $dropbox,
        SmdStatusMapper $mapper,
    ): void {
        $path = $entry['path_lower'];

        try {
            $json = $dropbox->downloadJson($path);

            if (! $json || ! isset($json['_id'])) {
                Log::warning('[Dropbox Webhook Job] JSON sin _id', ['path' => $path]);
                return;
            }

            $externalId   = $json['_id'];
            $isArchived   = $json['archived'] ?? false;
            $status       = $json['status']   ?? '';
            $shouldCancel = $isArchived || $mapper->isCancelled($status);

            $existing = DB::table('smd_synced_events')
                ->where('external_id', $externalId)
                ->first();

            if ($shouldCancel) {
                if ($existing) {
                    $incoming->cancelDropbox($externalId, $existing);
                    DB::table('smd_synced_events')
                        ->where('external_id', $externalId)
                        ->update(['status' => $status, 'archived_at' => now(), 'updated_at' => now()]);
                }
                return;
            }

            if ($existing) {
                $newHash = md5(json_encode($json));
                if (md5($existing->raw_payload ?? '') === $newHash) {
                    return;  // sin cambios
                }
                $incoming->updateDropbox($json, $existing);
                DB::table('smd_synced_events')
                    ->where('external_id', $externalId)
                    ->update(['raw_payload' => json_encode($json), 'status' => $status, 'updated_at' => now()]);
            } else {
                $result = $incoming->processDropbox($json);
                DB::table('smd_synced_events')->insert([
                    'external_id'  => $externalId,
                    'source_file'  => $path,
                    'activity_id'  => $result['activity_id'] ?? null,
                    'lead_id'      => $result['lead_id']      ?? null,
                    'raw_payload'  => json_encode($json),
                    'status'       => $status,
                    'processed_at' => now(),
                    'created_at'   => now(),
                    'updated_at'   => now(),
                ]);
            }

        } catch (\Throwable $e) {
            Log::error('[Dropbox Webhook Job] Error procesando archivo', [
                'path'  => $path,
                'error' => $e->getMessage(),
            ]);
        }
    }

    private function getCursor(): ?string
    {
        $row = DB::table('od_settings')->where('key', 'dropbox_cursor')->first();
        return $row?->value;
    }

    private function saveCursor(string $cursor): void
    {
        DB::table('od_settings')->updateOrInsert(
            ['key' => 'dropbox_cursor'],
            ['value' => $cursor, 'updated_at' => now()]
        );
    }
}
```

## Cambios en `DropboxService`

Agregar dos métodos a `packages/Webkul/Admin/src/Services/DropboxService.php`:

### `getInitialCursor(string $path): ?string`

Llama a `files/list_folder` para obtener un cursor limpio **sin** procesar los archivos listados. Solo captura el cursor final.

```php
public function getInitialCursor(string $path): ?string
{
    try {
        // Llamar list_folder y paginar hasta has_more = false para capturar el cursor final
        $cursor = null;
        $url    = 'https://api.dropboxapi.com/2/files/list_folder';
        $body   = ['path' => rtrim($path, '/'), 'recursive' => true];

        do {
            if ($cursor) {
                $url  = 'https://api.dropboxapi.com/2/files/list_folder/continue';
                $body = ['cursor' => $cursor];
            }

            $response = Http::withToken($this->token)
                ->withHeaders(['Content-Type' => 'application/json'])
                ->post($url, $body);

            if (! $response->successful()) {
                Log::warning('[Dropbox] Error obteniendo cursor inicial', ['status' => $response->status()]);
                return null;
            }

            $data   = $response->json();
            $cursor = $data['cursor'] ?? null;
            $more   = $data['has_more'] ?? false;

        } while ($more);

        return $cursor;

    } catch (\Throwable $e) {
        Log::error('[Dropbox] Excepcion obteniendo cursor inicial', ['error' => $e->getMessage()]);
        return null;
    }
}
```

### `listChanges(string $cursor): ?array`

Llama a `files/list_folder/continue` con el cursor guardado. Devuelve `['entries' => [...], 'cursor' => '...']` o `null` si hay error.

```php
public function listChanges(string $cursor): ?array
{
    try {
        $allEntries = [];
        $newCursor  = $cursor;

        do {
            $response = Http::withToken($this->token)
                ->withHeaders(['Content-Type' => 'application/json'])
                ->post('https://api.dropboxapi.com/2/files/list_folder/continue', [
                    'cursor' => $newCursor,
                ]);

            if (! $response->successful()) {
                // Cursor expirado (error 410 o similar) — resetear
                if ($response->status() === 409) {
                    Log::warning('[Dropbox] Cursor expirado, se requiere reinicialización');
                    DB::table('od_settings')->where('key', 'dropbox_cursor')->delete();
                }
                Log::warning('[Dropbox] Error en list_folder/continue', [
                    'status' => $response->status(),
                    'body'   => $response->body(),
                ]);
                return null;
            }

            $data       = $response->json();
            $newCursor  = $data['cursor']   ?? $newCursor;
            $more       = $data['has_more'] ?? false;
            $allEntries = array_merge($allEntries, $data['entries'] ?? []);

        } while ($more);

        return ['entries' => $allEntries, 'cursor' => $newCursor];

    } catch (\Throwable $e) {
        Log::error('[Dropbox] Excepcion en listChanges', ['error' => $e->getMessage()]);
        return null;
    }
}
```

## Nota sobre `QUEUE_CONNECTION`

Si `QUEUE_CONNECTION=sync` en `.env`, el job se ejecuta en la misma request HTTP, lo que podría superar el timeout de 10 segundos de Dropbox. En producción debe ser `database` o `redis`.

Si el entorno usa `sync`, el job igual funciona pero no hay garantía de respuesta rápida. Documentar esto en el `README` o en el `.env.example`.
